cmake_minimum_required(VERSION 3.22)
project(hypr_external C)

set(archive "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lua-5.4.9.tar.gz")
file(SHA256 "${archive}" archive_digest)
if(NOT archive_digest STREQUAL "2335b6c582a52654f94612bf10d2f4672805d05329aa6568b1d8cd9e5c6fb8e6")
    message(FATAL_ERROR "Unexpected Lua source digest")
endif()
file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
set(lua "${CMAKE_CURRENT_BINARY_DIR}/lua-5.4.9/src")
set(core lapi lcode lctype ldebug ldo ldump lfunc lgc llex lmem lobject lopcodes
    lparser lstate lstring ltable ltm lundump lvm lzio lauxlib)
list(TRANSFORM core PREPEND "${lua}/")
list(TRANSFORM core APPEND ".c")
add_library(hypr_external SHARED runtime.c ${core})
target_include_directories(hypr_external PRIVATE "${lua}")
target_compile_features(hypr_external PRIVATE c_std_11)
target_compile_options(hypr_external PRIVATE -fvisibility=hidden -fstack-protector-strong)
target_link_options(hypr_external PRIVATE -Wl,-z,relro,-z,now,-z,max-page-size=16384)
target_link_libraries(hypr_external PRIVATE m)

option(HYPR_RUNTIME_TEST_FIXTURES "Native worker fixtures for emulator instrumentation" OFF)
if(HYPR_RUNTIME_TEST_FIXTURES)
    target_compile_definitions(hypr_external PRIVATE HYPR_RUNTIME_TEST_FIXTURES)
    target_link_libraries(hypr_external PRIVATE log)
endif()
